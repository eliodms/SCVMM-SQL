LatestInventoryFolder

let
    Source = Folder.Files("\\AZURE-FS-LIB\\Workspaces\\Scripts\\Secure-Boot\\HyperV-SecureBoot-Inventory"),
    KeepRelevant = Table.SelectColumns(Source, {"Folder Path", "Name", "Date modified", "Content"}),
    FilterCsv = Table.SelectRows(KeepRelevant, each [Extension]? <> null or true),
    SplitFolder = Table.AddColumn(FilterCsv, "RunFolder", each List.Last(Text.Split(Text.TrimEnd([Folder Path], "\\"), "\\")), type text),
    GroupByRunFolder = Table.Group(
        SplitFolder,
        {"RunFolder", "Folder Path"},
        {{"LatestModified", each List.Max([Date modified]), type datetime}}
    ),
    SortDesc = Table.Sort(GroupByRunFolder, {{"RunFolder", Order.Descending}}),
    KeepTop = Table.FirstN(SortDesc, 1)
in
    KeepTop
